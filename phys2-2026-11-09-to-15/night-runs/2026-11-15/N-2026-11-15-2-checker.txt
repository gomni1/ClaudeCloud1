import numpy as np
D1,D2=10000,2000
def mu(q): return 1-q/2 if q<=1 else 1/(2*q)
def c(s,l): return np.maximum(0,1-s/l)
def Nform(D,l): return (D1+D2)*(1-mu(D/l))
def Nsum(D,l):  # direct continuous-position sum
    n=np.arange(D1)+0.5; m=np.arange(D2)+0.5
    return np.sum(1-c(D*n/D1,l))+np.sum(1-c(D*(1-m/D2),l))
def Nlat(y1,y2,x,l): # lattice: route reads grain round(pos); integer-distance correlation
    n=np.arange(D1)+0.5; m=np.arange(D2)+0.5
    k1=np.abs(np.round(y1*n/D1)-np.round(y2*n/D1))
    k2=np.abs(np.round(y1+(x-y1)*m/D2)-np.round(y2+(x-y2)*m/D2))
    return np.sum(1-c(k1,l))+np.sum(1-c(k2,l))
print("R3 table: l, Delta, formula, direct sum, lattice-rounded (y=-D/2..,x=37 / slit pair)")
for l in (1,5,100,1000):
    for D in (1,4,100,200):
        y1,y2=((-100,-100+D) if D<100 else ((-100,0) if D==100 else (-100,100)))
        print(l,D,round(Nform(D,l),2),round(Nsum(D,l),2),round(Nlat(y1,y2,37,l),2),round(Nlat(y1,y2,-300,l),2))
print("limits: l=1e-9 ->",Nform(100,1e-9)," l=1e12 ->",Nform(100,1e12))
# leg split: fraction of N_un from leg 2 is always D2/(D1+D2)
print("leg split check (D=4,l=5): leg1",np.sum(1-c(4*(np.arange(D1)+.5)/D1,5)),"leg2",np.sum(1-c(4*(1-(np.arange(D2)+.5)/D2),5)))
# smooth correlation: c=exp(-(s/l)^2) small d -> d^2/(3 l^2)
for d in (5,10,20):
    l=1000;n=np.arange(D1)+.5;m=np.arange(D2)+.5
    N=np.sum(1-np.exp(-(d*n/D1/l)**2))+np.sum(1-np.exp(-(d*(1-m/D2)/l)**2))
    print("gaussian corr d",d,"N_un",N,"pred (D1+D2)d^2/(3l^2)",12000*d*d/3/l/l)
# Monte Carlo one pair
rng=np.random.default_rng(1)
def mc(D,l,sig,T=4000):
    n=np.arange(D1)+0.5; m=np.arange(D2)+0.5
    cc=np.concatenate([c(D*n/D1,l),c(D*(1-m/D2),l)])
    acc=[]
    for _ in range(T):
        a=rng.standard_normal(cc.size); b=cc*a+np.sqrt(1-cc**2)*rng.standard_normal(cc.size)
        acc.append(np.cos(2*sig*(a.sum()-b.sum())))
    acc=np.array(acc); return acc.mean(),acc.std()/np.sqrt(T)
print("MC one pair: Delta,l,sigma, MC mean +- se, exp(-4 s^2 N_un)")
for D,l,s in ((100,100,0.005),(4,5,0.006),(200,1,0.004),(100,1000,0.02)):
