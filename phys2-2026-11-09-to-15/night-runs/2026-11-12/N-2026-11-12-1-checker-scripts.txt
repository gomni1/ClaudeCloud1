def par(d):
    e = D_[d]; return list(e['k']), int(e['m']), int(e['kr'])
def make(d):
    (k1,k2,k3,k5,k6,k7,k8,k10), m, kr = par(d)
    lb = betaln(kr, m-kr+1)
    def rho(x):
        x = min(max(x,0.0),1.0); return k3*betainc(kr, m-kr+1, x)
    def drho(x):
        x = min(max(x,1e-300),1.0); return k3*np.exp((kr-1)*np.log(x)+(m-kr)*np.log1p(-x) - lb) if x<1 else 0.0
    # reduced coords y=(C,L,P), V=1-C-L-P
    def f(t,y):
        C,L,P = y; V = 1-C-L-P; r = k10+rho(C)
        return [-(k1+k2+k7)*C + k6*L + r*V, k1*C-(k5+k6)*L, k7*C-k8*P]
    def J(t,y):
        C,L,P = y; V = 1-C-L-P; r = k10+rho(C)
        return np.array([[-(k1+k2+k7)+drho(C)*V-r, k6-r, -r],[k1,-(k5+k6),0.0],[k7,0.0,-k8]])
    return f,J,rho,drho

def analyse(d, y0, method, nper=14, rtol=1e-10, atol=1e-13):
    f,J,_,_ = make(d); Tg = CLAIM[d]   # claimed period used ONLY to set integration length
    sol = solve_ivp(f,(0,nper*Tg),y0,method=method,jac=J,rtol=rtol,atol=atol,dense_output=True)
    if not sol.success: return dict(ok=False,msg=sol.message)
    t0 = (nper-5)*Tg; mask = sol.t>=t0
    tt = sol.t[mask]; C = sol.y[0][mask]
    # refine extrema using dense grid
    tg = np.linspace(t0, sol.t[-1], 400001); Y = sol.sol(tg); Cg = Y[0]
    cmin, cmax = Cg.min(), Cg.max(); mid = 0.5*(cmin+cmax)
    if cmax-cmin < 1e-6: return dict(ok=True, rest=True, y=sol.y[:,-1], cmin=cmin, cmax=cmax)
    s = Cg-mid; idx = np.where((s[:-1]<0)&(s[1:]>=0))[0]
    tc = np.array([brentq(lambda t: sol.sol(t)[0]-mid, tg[i], tg[i+1], xtol=1e-13) for i in idx])
    per = np.diff(tc); sec = np.array([sol.sol(t) for t in tc])
    frac = np.mean(Cg[(tg>=tc[0])&(tg<tc[-1])]>mid) if len(tc)>1 else np.nan
    V = 1-Y.sum(0)
    return dict(ok=True, rest=False, per=per, sec=sec, cmin=cmin, cmax=cmax, frac=frac, nsteps=len(sol.t),
                Lr=(Y[1].min(),Y[1].max()), Pr=(Y[2].min(),Y[2].max()), Vr=(V.min(),V.max()))
